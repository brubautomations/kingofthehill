exports.handler = async (event) => {
  const GAS = 'https://script.google.com/macros/s/AKfycbyjFDcDuP-WrMtzHZhPu_p9KCnrXMv3mCIDE5aB737h6KfhuWsFHXukiWZtt96dlqVI/exec';
  try {
    const res = await fetch(GAS, {
      method: event.httpMethod,
      headers: { 'Content-Type': 'application/json' },
      body: event.body || null,
      redirect: 'follow'
    });
    const body = await res.text();
    return { statusCode: 200, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }, body };
  } catch(e) {
    return { statusCode: 500, headers: { 'Access-Control-Allow-Origin': '*' }, body: JSON.stringify({ success: false, error: e.message }) };
  }
};
